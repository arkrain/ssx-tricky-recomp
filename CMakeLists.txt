add_library(ssx_vertex STATIC src/nv2a_vertex.c
    ${VSH_DIR}/nv2a_vsh_cpu.c ${VSH_DIR}/nv2a_vsh_disassembler.c
    ${VSH_DIR}/nv2a_vsh_emulator.c ${VSH_DIR}/nv2a_vsh_emulator_execution_state.c)
target_include_directories(ssx_vertex PRIVATE "${VSH_DIR}" PUBLIC src)
option(SSX_NATIVE_VERTEX "Compile embedded guest vertex programs ahead of time" ON)
set(SSX_GAME_XBE "${CMAKE_CURRENT_SOURCE_DIR}/game/default.xbe" CACHE FILEPATH "Original XBE for vertex compilation")
set(SSX_NATIVE_VERTEX_AVAILABLE OFF)
if(SSX_NATIVE_VERTEX AND (NOT CMAKE_CROSSCOMPILING OR CMAKE_HOST_UNIX))
    set(SSX_NATIVE_VERTEX_AVAILABLE ON)
    if(CMAKE_CROSSCOMPILING)
        # This generator must run on the build host, not the target platform.
        find_program(SSX_VERTEX_HOST_CC NAMES cc gcc clang NO_CMAKE_FIND_ROOT_PATH REQUIRED)
        set(VSH_COMPILER "${CMAKE_CURRENT_BINARY_DIR}/compile_vertex_programs_host")
        add_custom_command(OUTPUT "${VSH_COMPILER}"
            COMMAND "${SSX_VERTEX_HOST_CC}" -std=c11 -O2
                -I "${VSH_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/compile_vertex_programs.c"
                "${VSH_DIR}/nv2a_vsh_disassembler.c" -o "${VSH_COMPILER}"
            DEPENDS tools/compile_vertex_programs.c "${VSH_DIR}/nv2a_vsh_disassembler.c"
                "${VSH_DIR}/nv2a_vsh_disassembler.h" VERBATIM)
        set(VSH_COMPILER_DEP "${VSH_COMPILER}")
    else()
        add_executable(compile_vertex_programs tools/compile_vertex_programs.c ${VSH_DIR}/nv2a_vsh_disassembler.c)
        target_include_directories(compile_vertex_programs PRIVATE "${VSH_DIR}")
        set(VSH_COMPILER "$<TARGET_FILE:compile_vertex_programs>")
        set(VSH_COMPILER_DEP compile_vertex_programs)
    endif()
    set(NATIVE_VSH "${CMAKE_CURRENT_BINARY_DIR}/nv2a_native_programs.inc")
    add_custom_command(OUTPUT "${NATIVE_VSH}"
        COMMAND "${VSH_COMPILER}" "${SSX_GAME_XBE}" "${NATIVE_VSH}"
        DEPENDS ${VSH_COMPILER_DEP} "${SSX_GAME_XBE}" VERBATIM)
    file(READ "${VSH_DIR}/nv2a_vsh_cpu.c" NATIVE_OPS)
    # Keep each opcode inside its compiled program so the optimizer can remove
    # temporary register arrays. Preserve the interpreter's arithmetic exactly.
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        set(NATIVE_OP_INLINE "static inline __attribute__((always_inline))")
    else()
        set(NATIVE_OP_INLINE "static inline")
    endif()
    string(REPLACE "void nv2a_vsh_cpu_" "${NATIVE_OP_INLINE} void ssx_native_op_" NATIVE_OPS "${NATIVE_OPS}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/nv2a_native_ops.inc" "${NATIVE_OPS}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${VSH_DIR}/nv2a_vsh_cpu.c")
    target_sources(ssx_vertex PRIVATE src/nv2a_vertex_native.c "${NATIVE_VSH}")
    target_include_directories(ssx_vertex PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_definitions(ssx_vertex PRIVATE SSX_NATIVE_VERTEX)
endif()
target_link_libraries(xbox_kernel PUBLIC ssx_vertex)
add_library(ssx_gl STATIC src/nv2a_gl.c src/nv2a_gpu_geometry.c)

            scripts/generate-vertex-glsl-probe.py src/nv2a_gpu_ops.glsl src/nv2a_gpu_clip.glsl
        VERBATIM)
    target_sources(ssx_gl PRIVATE "${GPU_VSH}")
    target_compile_definitions(ssx_gl PRIVATE SSX_GPU_GENERATED)
endif()

    string(REPLACE "\n" "\\n\"\n\"" GL_FRAGMENT "${GL_FRAGMENT}")

    target_compile_options(ssx_gl PRIVATE -O2)
endif()

        find_package(PkgConfig REQUIRED)
    endif()

    if(SSX_AV_FOUND)
        target_compile_definitions(ssx_movies PRIVATE SSX_HAVE_FFMPEG)

            target_link_libraries(ssx_movies PRIVATE OpenSSL::SSL)
        endif()
    elseif(WIN32)
        message(FATAL_ERROR "SSX_WINDOWS_MOVIES requires target FFmpeg libraries")
    else()
        message(STATUS "FFmpeg development files unavailable: startup movies require --skip-intro")
    endif()
endif()
